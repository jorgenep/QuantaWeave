# Base data and smoke-test environment
uv venv --python 3.12 .venv
source .venv/bin/activate
uv pip install datasets transformers tqdm huggingface_hub
deactivate

# Axolotl environment: use Python 3.12, not Python 3.14.
uv venv --python 3.12 .axolotl-venv
source .axolotl-venv/bin/activate
uv pip install -e './axolotl[deepspeed]'
deactivate
